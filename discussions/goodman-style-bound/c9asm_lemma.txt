/-- The `C₉` necklace identity, unrolled from `ccomp_necklace 7`. -/
lemma ccomp9_necklace (hU : IsGraphon U μ) :
    tr μ (Kpow μ (Wk U) 8)
      = mean μ (vcomp U μ 8) - ip μ (pathFun U μ 1) (vcomp U μ 7)
        + ip μ (pathFun U μ 2) (vcomp U μ 6) - ip μ (pathFun U μ 3) (vcomp U μ 5)
        + ip μ (pathFun U μ 4) (vcomp U μ 4) - ip μ (pathFun U μ 5) (vcomp U μ 3)
        + ip μ (pathFun U μ 6) (vcomp U μ 2) - ip μ (pathFun U μ 7) (vcomp U μ 1)
        + xden U μ 8 - tr μ (Kpow μ U 8) := by
  have h := ccomp_necklace hU 7
  simp only [Finset.sum_range_succ, Finset.sum_range_zero, ip_pathFun_zero] at h
  norm_num at h
  rw [h]; ring
